Python Code Samples
basic_analysis.py 
import os
from azure.ai.contentsafety import ContentSafetyClient
from azure.core.credentials import AzureKeyCredential
from azure.ai.contentsafety.models import AnalyzeTextOptions

# ── Setup ──────────────────────────────────────────────
endpoint = os.environ["CONTENT_SAFETY_ENDPOINT"]
key      = os.environ["CONTENT_SAFETY_KEY"]

client = ContentSafetyClient(
    endpoint,
    AzureKeyCredential(key)
)

# ── Analyze ────────────────────────────────────────────
text = "I want to hurt someone today."

response = client.analyze_text(
    AnalyzeTextOptions(text=text)
)

# ── Read severity scores ────────────────────────────────
for category in response.categories_analysis:
    print(f"[{category.category}] severity = {category.severity}")

# Output example:
# [Hate]     severity = 0
# [Violence] severity = 4
# [Sexual]   severity = 0
# [SelfHarm] severity = 2
threshold_check.py 
from azure.ai.contentsafety.models import (
    AnalyzeTextOptions,
    TextCategory,
)

# Severity threshold constants — mirror the SDK enum values
THRESHOLD = {
    "off":    None,   # never block
    "low":    2,      # block severity >= 2
    "medium": 4,      # block severity >= 4 (default)
    "high":   6,      # block severity >= 6 only
}

def should_block(severity: int, level: str = "medium") -> bool:
    t = THRESHOLD.get(level)
    if t is None:
        return False
    return severity >= t

# ── Demo ──────────────────────────────────────────────
samples = [
    ("Hello, how are you?", 0),
    ("You're an idiot.",    2),
    ("I will find you.",    4),
    ("Graphic violence…",   6),
]

for text, sev in samples:
    result = {
        lvl: should_block(sev, lvl)
        for lvl in THRESHOLD
    }
    print(f"sev={sev} | {result}")

# sev=0 | {'off': False, 'low': False, 'medium': False, 'high': False}
# sev=2 | {'off': False, 'low': True,  'medium': False, 'high': False}
# sev=4 | {'off': False, 'low': True,  'medium': True,  'high': False}
# sev=6 | {'off': False, 'low': True,  'medium': True,  'high': True }
multi_category.py 
from azure.ai.contentsafety.models import (
    AnalyzeTextOptions, TextCategory
)

# Per-category thresholds — fine-grained control
CATEGORY_THRESHOLDS = {
    TextCategory.HATE:      2,  # strict on hate speech
    TextCategory.VIOLENCE:  4,  # moderate on violence
    TextCategory.SEXUAL:    6,  # lenient (adult platform)
    TextCategory.SELF_HARM: 2,  # strict on self-harm
}

def evaluate_content(client, text: str) -> dict:
    response = client.analyze_text(
        AnalyzeTextOptions(text=text)
    )

    violations = []
    for item in response.categories_analysis:
        threshold = CATEGORY_THRESHOLDS.get(item.category, 4)
        if item.severity >= threshold:
            violations.append({
                "category":  item.category,
                "severity":  item.severity,
                "threshold": threshold,
                "blocked":   True,
            })

    return {
        "text":     text,
        "blocked": bool(violations),
        "reasons": violations,
    }

# ── Usage ─────────────────────────────────────────────
result = evaluate_content(client, "Some test input here")
print(result)
safety_guard.py 
from dataclasses import dataclass, field
from typing import Literal
from azure.ai.contentsafety.models import AnalyzeTextOptions

PolicyLevel = Literal["strict", "standard", "permissive"]

# Map policy presets to per-category thresholds
POLICIES: dict[PolicyLevel, dict[str, int]] = {
    "strict": {           # children / education
        "Hate": 2, "Violence": 2,
        "Sexual": 2, "SelfHarm": 2,
    },
    "standard": {         # general consumer apps
        "Hate": 2, "Violence": 4,
        "Sexual": 4, "SelfHarm": 2,
    },
    "permissive": {       # adult / professional
        "Hate": 4, "Violence": 6,
        "Sexual": 6, "SelfHarm": 4,
    },
}

@dataclass
class SafetyResult:
    blocked: bool
    scores: dict[str, int] = field(default_factory=dict)
    violations: list[str] = field(default_factory=list)

def check_safety(
    client,
    text: str,
    policy: PolicyLevel = "standard"
) -> SafetyResult:
    thresholds = POLICIES[policy]
    resp = client.analyze_text(AnalyzeTextOptions(text=text))

    scores      = {}
    violations  = []

    for item in resp.categories_analysis:
        cat = item.category
        sev = item.severity
        scores[cat] = sev
        if sev >= thresholds.get(cat, 4):
            violations.append(cat)

    return SafetyResult(
        blocked    = bool(violations),
        scores     = scores,
        violations = violations,
    )

# ── Usage ─────────────────────────────────────────────
r = check_safety(client, "Some harmful text", policy="standard")
print(f"Blocked: {r.blocked}")
print(f"Violations: {r.violations}")
print(f"Scores: {r.scores}")

Best Practices
🎯
Per-Category Thresholds
Don't use a single global threshold. Set tighter thresholds on SelfHarm and Hate, and looser ones on Violence for gaming/news contexts.
📊
Log Everything
Store raw severity scores alongside your block decisions. Use this data to tune thresholds over time — what's too strict today may be right in a month.
⚡
Async Client
Use ContentSafetyClient's async variant for high-throughput pipelines. Wrap calls with asyncio.gather() to batch requests without SDK-level batching.
🔄
Severity = 0 ≠ Guaranteed Safe
A score of 0 means no detectable harm at that confidence level, not a guarantee. Layer with blocklists and custom categories for defense-in-depth.
